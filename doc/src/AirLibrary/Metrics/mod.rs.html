1//! # Metrics Collection and Export Module
2//!
3//! Provides Prometheus-compatible metrics collection and export for the Air
4//! daemon. Includes request latency histograms, resource usage metrics, error
5//! rate tracking, thread-safe metric updates with overflow protection, and
6//! aggregation validation.
7//!
8//! ## Responsibilities
9//!
10//! ### Request Metrics
11//! - Total request count tracking with counter overflow protection
12//! - Success/failure rate calculation
13//! - Latency histogram with bucket aggregation
14//! - Request throughput measurement
15//!
16//! ### Error Metrics
17//! - Total error count with classification
18//! - Error rate calculation by type
19//! - Error aggregation and validation
20//! - Error threshold alerting
21//!
22//! ### Resource Metrics
23//! - Memory usage monitoring with allocation tracking
24//! - CPU utilization percentage
25//! - Active connection count
26//! - Thread activity monitoring
27//! - Resource pool utilization
28//!
29//! ### Service-Specific Metrics
30//! - Authentication success/failure tracking
31//! - Download completion rates
32//! - Indexing operation metrics
33//! - Update deployment statistics
34//!
35//! ## Integration with Mountain
36//!
37//! Metrics flow directly to Mountain's telemetry UI:
38//! - Prometheus exposition format endpoint
39//! - Real-time metric streaming
40//! - Historical data retention
41//! - Custom dashboards and alerting
42//!
43//! ## VSCode Telemetry References
44//!
45//! Similar telemetry patterns used in VSCode for:
46//! - Performance monitoring and profiling
47//! - Usage statistics and feature adoption
48//! - Error tracking and crash reporting
49//! - Extension marketplace analytics
50//!
51//! Reference:
52//! vs/workbench/services/telemetry
53//!
54//! ## Thread Safety
55//!
56//! All metric updates are thread-safe using:
57//! - Arc for shared ownership across threads
58//! - Atomic operations where possible
59//! - Mutex locks for complex aggregations
60//! - Lock-free counters for high-frequency updates
61//!

92/// Overflow-protected metric update helper
93struct MetricGuard {
94	current:u64,
95	max:u64,
96}
97
98impl MetricGuard {
99	fn new(current:u64, max:u64) -> Self { Self { current, max } }
100
101	/// Increment with overflow protection
102	fn increment(&mut self) -> bool {
103		if self.current < self.max.saturating_sub(1) {
104			self.current += 1;
105			true
106		} else {
107			warn!("[Metrics] Metric overflow detected, wrapping around");
108			self.current = 0;
109			true
110		}
111	}
112}

114/// Aggregation validation for metric integrity
115#[derive(Debug)]
116struct AggregationValidator {
117	last_timestamp:Instant,
118	validation_window:Duration,
119}
120
121impl AggregationValidator {
122	fn new(validation_window_secs:u64) -> Self {
123		Self {
124			last_timestamp:Instant::now(),
125			validation_window:Duration::from_secs(validation_window_secs),
126		}
127	}
128
129	/// Validate aggregation is within time window
130	fn validate(&mut self) -> std::result::Result<(), String> {
131		let now = Instant::now();
132		if now.duration_since(self.last_timestamp) > self.validation_window {
133			warn!("[Metrics] Aggregation outside validation window, resetting");
134			self.last_timestamp = now;
135			Ok(())
136		} else {
137			Ok(())
138		}
139	}
140}

214	/// Validate aggregation window before metric updates
215	pub fn ValidateAggregation(&self) -> Result<()> {
216		match self.aggregator.lock() {
217			Ok(mut validator) => validator.validate().map_err(|e| AirError::Internal(e)),
218			Err(_) => {
219				warn!("[Metrics] Failed to acquire aggregation validator lock");
220				Ok(())
221			},
222		}
223	}

225	/// Record a successful request with thread-safe atomic updates
226	pub fn RecordRequestSuccess(&self, LatencySeconds:f64) {
227		self.ValidateAggregation();
228
229		let LatencyMs = (LatencySeconds * 1000.0) as u64;
230
231		// Update request counts
232		let _ = self.requests_total.fetch_add(1, Ordering::Relaxed);
233		let _ = self.requests_successful.fetch_add(1, Ordering::Relaxed);
234
235		// Update latency metrics
236		let _ = self.request_latency_sum_ms.fetch_add(LatencyMs, Ordering::Relaxed);
237		let _ = self.request_latency_count.fetch_add(1, Ordering::Relaxed);
238
239		// Update min/max latency
240		MinMaxUpdate(&self.request_latency_min_ms, &self.request_latency_max_ms, LatencyMs);
241
242		debug!("[Metrics] Recorded successful request with latency: {:.3}s", LatencySeconds);
243	}

245	/// Record a failed request with thread-safe atomic updates
246	pub fn RecordRequestFailure(&self, ErrorType:&str, LatencySeconds:f64) {
247		self.ValidateAggregation();
248
249		let LatencyMs = (LatencySeconds * 1000.0) as u64;
250
251		// Update request counts
252		let _ = self.requests_total.fetch_add(1, Ordering::Relaxed);
253		let _ = self.requests_failed.fetch_add(1, Ordering::Relaxed);
254		let _ = self.errors_total.fetch_add(1, Ordering::Relaxed);
255
256		// Update latency metrics
257		let _ = self.request_latency_sum_ms.fetch_add(LatencyMs, Ordering::Relaxed);
258		let _ = self.request_latency_count.fetch_add(1, Ordering::Relaxed);
259
260		// Update min/max latency
261		MinMaxUpdate(&self.request_latency_min_ms, &self.request_latency_max_ms, LatencyMs);
262
263		// Track error by type with redaction
264		let RedactedError = self.RedactErrorType(ErrorType);
265		let RedactedErrorClone = RedactedError.clone();
266		if let Ok(mut error_map) = self.errors_by_type.lock() {
267			*error_map.entry(RedactedError).or_insert(0) += 1;
268		}
269
270		debug!(
271			"[Metrics] Recorded failed request: {}, latency: {:.3}s",
272			RedactedErrorClone, LatencySeconds
273		);
274	}

484/// Helper function to update min/max values atomically
485fn MinMaxUpdate(MinMetric:&AtomicU64, MaxMetric:&AtomicU64, Value:u64) {
486	let mut CurrentMin = MinMetric.load(Ordering::Relaxed);
487	let mut CurrentMax = MaxMetric.load(Ordering::Relaxed);
488
489	loop {
490		if Value < CurrentMin {
491			match MinMetric.compare_exchange_weak(CurrentMin, Value, Ordering::Relaxed, Ordering::Relaxed) {
492				Ok(_) => break,
493				Err(NewMin) => CurrentMin = NewMin,
494			}
495		} else if Value > CurrentMax {
496			match MaxMetric.compare_exchange_weak(CurrentMax, Value, Ordering::Relaxed, Ordering::Relaxed) {
497				Ok(_) => break,
498				Err(NewMax) => CurrentMax = NewMax,
499			}
500		} else {
501			break;
502		}
503	}
504}

537impl MetricsData {
538	/// Calculate success rate as percentage
539	pub fn SuccessRate(&self) -> f64 {
540		if self.requests_total == 0 {
541			return 100.0;
542		}
543		(self.requests_successful as f64 / self.requests_total as f64) * 100.0
544	}
545
546	/// Calculate download success rate
547	pub fn DownloadSuccessRate(&self) -> f64 {
548		if self.downloads_total == 0 {
549			return 100.0;
550		}
551		(self.downloads_completed as f64 / self.downloads_total as f64) * 100.0
552	}
553
554	/// Calculate error rate
555	pub fn ErrorRate(&self) -> f64 {
556		if self.requests_total == 0 {
557			return 0.0;
558		}
559		(self.errors_total as f64 / self.requests_total as f64) * 100.0
560	}
561}

563/// Global metrics collector instance
564static METRICS_INSTANCE:std::sync::OnceLock<MetricsCollector> = std::sync::OnceLock::new();
565
566/// Get or initialize the global metrics collector
567pub fn GetMetrics() -> &'static MetricsCollector { METRICS_INSTANCE.get_or_init(|| MetricsCollector::default()) }
568
569/// Initialize the global metrics collector
570pub fn InitializeMetrics() -> Result<()> {
571	let _collector = GetMetrics();
572	info!("[Metrics] Global metrics collector initialized");
573	Ok(())
