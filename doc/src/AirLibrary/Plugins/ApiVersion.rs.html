1//! API version compatibility management for the plugin system.
2//!
3//! `ApiVersion` represents a semver triple with an optional pre-release tag.
4//! Compatibility follows the semver major-version rule: plugins with the same
5//! major version and a minor version ≥ the host's are considered compatible.
6//!
7//! `ApiVersionManager` tracks the current host version and the set of
8//! explicitly registered compatible versions.
9
10use serde::{Deserialize, Serialize};
11
12use crate::Result;
13
14/// Semver-style API version.
15#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
16pub struct ApiVersion {
17	pub major:u32,
18
19	pub minor:u32,
20
21	pub patch:u32,
22
23	pub PreRelease:Option<String>,
24}
25
26impl ApiVersion {
27	/// The current host API version.
28	pub fn current() -> Self { Self { major:1, minor:0, patch:0, PreRelease:None } }
29
30	/// Parse from `"major.minor.patch"` or `"major.minor.patch.pre"`.
31	pub fn parse(version:&str) -> Result<Self> {
32		let parts:Vec<&str> = version.split('.').collect();
33
34		if parts.len() < 3 {
35			return Err(crate::AirError::Plugin("Invalid version format".to_string()));
36		}
37
38		Ok(Self {
39			major:parts[0]
40				.parse()
41				.map_err(|_| crate::AirError::Plugin("Invalid major version".to_string()))?,
42			minor:parts[1]
43				.parse()
44				.map_err(|_| crate::AirError::Plugin("Invalid minor version".to_string()))?,
45			patch:parts[2]
46				.parse()
47				.map_err(|_| crate::AirError::Plugin("Invalid patch version".to_string()))?,
48			PreRelease:parts.get(3).map(|s| s.to_string()),
49		})
50	}
51
52	/// `true` when `other` is compatible: same major, minor ≥ self.minor.
53	pub fn IsCompatible(&self, other:&ApiVersion) -> bool { self.major == other.major && other.minor >= self.minor }
54}
55
56/// Tracks the current host API version and a set of compatible peer versions.
57pub struct ApiVersionManager {
58	CurrentVersion:ApiVersion,
59
60	CompatibleVersions:Vec<ApiVersion>,
61}
62
63impl ApiVersionManager {
64	pub fn new() -> Self {
65		let current = ApiVersion::current();
66
67		Self { CompatibleVersions:vec![current.clone()], CurrentVersion:current }
68	}
69
70	pub fn current(&self) -> &ApiVersion { &self.CurrentVersion }
71
72	pub fn IsCompatible(&self, version:&ApiVersion) -> bool { self.CurrentVersion.IsCompatible(version) }
73
74	/// Register `version` as compatible if it passes the compatibility check.
75	pub fn register_compatible(&mut self, version:ApiVersion) {
76		if self.IsCompatible(&version) && !self.CompatibleVersions.contains(&version) {
77			self.CompatibleVersions.push(version);
78		}
79	}
80}
81
82impl Default for ApiVersionManager {
83	fn default() -> Self { Self::new() }
84}
